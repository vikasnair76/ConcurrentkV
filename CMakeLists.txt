cmake_minimum_required(VERSION 3.16)
project(ConcurrentKV VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_ASAN "Address and undefined behavior sanitizers" OFF)
option(ENABLE_TSAN "Thread sanitizer (separate build)" OFF)
if(ENABLE_ASAN AND ENABLE_TSAN)
  message(FATAL_ERROR "Use separate ASan and TSan builds")
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
if(ENABLE_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(ENABLE_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
add_library(kvcore src/write_ahead_log.cpp src/kv_store.cpp src/thread_pool.cpp src/command_parser.cpp src/metrics.cpp src/tcp_server.cpp)
target_include_directories(kvcore PUBLIC include)
target_link_libraries(kvcore PUBLIC Threads::Threads)
add_executable(kv_server src/main.cpp)
target_link_libraries(kv_server PRIVATE kvcore)
add_executable(kv_client client/kv_client.cpp)
target_link_libraries(kv_client PRIVATE kvcore)
add_executable(kv_bench benchmarks/load_test.cpp)
target_link_libraries(kv_bench PRIVATE kvcore)
enable_testing()
foreach(name store concurrency recovery network socket_io)
  add_executable(test_${name} tests/test_${name}.cpp)
  target_link_libraries(test_${name} PRIVATE kvcore)
  add_test(NAME ${name} COMMAND test_${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 60)
endforeach()

add_executable(test_process tests/test_process.cpp)
target_link_libraries(test_process PRIVATE kvcore)
add_test(NAME process COMMAND test_process $<TARGET_FILE:kv_server>)
set_tests_properties(process PROPERTIES TIMEOUT 90)
